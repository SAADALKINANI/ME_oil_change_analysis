import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from scipy import stats
import warnings
warnings.filterwarnings('ignore')


df = pd.read_csv('Middle_East_Economic_Data_1990_2024_with_Oil.csv')

oil_df = df.groupby('Year')['Brent_Oil_Price_USD_per_barrel'].first().reset_index()
oil_df.columns = ['Year', 'Oil_Price']
years      = oil_df['Year'].values
oil_prices = oil_df['Oil_Price'].values

#Create GDP-per-capita pivot table
gdp_pivot = df.pivot_table(
    index='Year', columns='Country', values='GDP_per_capita_current_USD'
).reset_index()

growth_pivot = df.pivot_table(
    index='Year', columns='Country', values='GDP_growth_annual_pct'
).reset_index()

combined = oil_df.merge(gdp_pivot, on='Year').merge(
    growth_pivot, on='Year', suffixes=('_gdp', '_growth')
)

growth_cols = [c for c in combined.columns if c.endswith('_growth')]
combined['Distress_Index'] = combined[growth_cols].apply(
    lambda row: (row < 0).sum() / max(row.notna().sum(), 1), axis=1
) * 10

print('Pearson correlations: Oil Price vs GDP per Capita')

for country in countries:
    col = f'{country}_gdp'
    sub = combined[['Oil_Price', col]].dropna()
    r, p = stats.pearsonr(sub['Oil_Price'], sub[col])
    print(country, ': r =', round(r, 3),'|  p =', round(p, 4))


print('\nExporter vs Non-Exporter oil sensitivity')
groups = {
    'Exporters': [
        'Saudi Arabia', 'Kuwait', 'United Arab Emirates',

    ],
    'Non-Exporters': [
        'Egypt', 'Jordan', 'Lebanon',
        'Yemen', 'Iran', 'Israel'
    ]
}

for label, countries in groups.items():
    sub = df[df['Country'].isin(countries)][
        ['Brent_Oil_Price_USD_per_barrel',
         'GDP_per_capita_current_USD']
    ].dropna()
    r, p = stats.pearsonr(
        sub['Brent_Oil_Price_USD_per_barrel'],
        sub['GDP_per_capita_current_USD']
    )
    print(label, ': r =', round(r, 3),' |  p =', round(p, 4))

fig, axes = plt.subplots(1, 3, figsize=(12, 4))

countries = ['Egypt', 'Kuwait', 'United Arab Emirates']
labels = ['Egypt', 'Kuwait', 'UAE']

for i, country in enumerate(countries):
    ax = axes[i]

    col = country + '_gdp'
    sub = combined[['Oil_Price', col]].dropna()

    x = sub['Oil_Price']
    y = sub[col] / 1000


    ax.scatter(x, y)

    m, b = np.polyfit(x, y, 1)
    ax.plot(x, m*x + b, linestyle='--')

    ax.set_title(labels[i])
    ax.set_xlabel('Oil Price')
    ax.set_ylabel('GDP per Capita ($000s)')

exporter_countries = [
    'Saudi Arabia', 'Kuwait', 'United Arab Emirates',
    'Qatar', 'Bahrain', 'Oman', 'Iraq'
]

non_exporter_countries = [
    'Egypt', 'Jordan', 'Lebanon',
    'Yemen', 'Iran', 'Israel'
]

fig, axes = plt.subplots(1, 2, figsize=(12, 5))

groups = [
    (exporter_countries, 'Oil Exporters'),
    (non_exporter_countries, 'Non-Exporters')
]

for i, (group, title) in enumerate(groups):
    ax = axes[i]

    for country in group:

        sub = df[df['Country'] == country][
            ['Brent_Oil_Price_USD_per_barrel',
             'GDP_per_capita_current_USD']
        ].dropna()

        x = sub['Brent_Oil_Price_USD_per_barrel']
        y = sub['GDP_per_capita_current_USD'] / 1000

        ax.scatter(x, y, alpha=0.5, label=country)

    ax.set_title(title)
    ax.set_xlabel('Oil Price')
    ax.set_ylabel('GDP per Capita ($000s)')

    ax.legend(fontsize=8)

plt.tight_layout()
plt.show()
